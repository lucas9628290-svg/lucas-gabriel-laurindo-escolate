/* ===== PARTE 2 - GALERIA ===== */ const inputFoto = document.getElementById("foto"); const grid = document.getElementById("grid"); let fotos = JSON.parse(localStorage.getItem("lucas_gallery")) || []; mostrarFotos(); inputFoto.addEventListener("change", function () { const arquivo = this.files[0]; if (!arquivo) return; const leitor = new FileReader(); leitor.onload = function (e) { fotos.unshift({ imagem: e.target.result, data: new Date().toLocaleString("pt-BR") }); salvarFotos(); mostrarFotos(); }; leitor.readAsDataURL(arquivo); }); function salvarFotos() { localStorage.setItem("lucas_gallery", JSON.stringify(fotos)); } function mostrarFotos() { grid.innerHTML = ""; if (fotos.length === 0) { grid.innerHTML = `
📷 Nenhuma foto enviada.
`; return; } fotos.forEach((foto, indice) => { const card = document.createElement("div"); card.className = "card"; card.innerHTML = `
${foto.data}
`; grid.appendChild(card); }); } function baixarImagem(src) { const a = document.createElement("a"); a.href = src; a.download = "MinhaFoto.png"; a.click(); } function apagarImagem(indice) { if (confirm("Deseja apagar esta foto?")) { fotos.splice(indice, 1); salvarFotos(); mostrarFotos(); } } /* ===== VISUALIZAR FOTO ===== */ const fundo = document.createElement("div"); fundo.style.cssText = ` display:none; position:fixed; top:0; left:0; width:100%; height:100%; background:rgba(0,0,0,.9); justify-content:center; align-items:center; z-index:9999; `; const imgGrande = document.createElement("img"); imgGrande.style.maxWidth = "90%"; imgGrande.style.maxHeight = "90%"; imgGrande.style.borderRadius = "15px"; fundo.appendChild(imgGrande); document.body.appendChild(fundo); function abrirImagem(src){ imgGrande.src = src; fundo.style.display = "flex"; } fundo.onclick = function(){ fundo.style.display = "none"; }; /* ===== DRAG & DROP ===== */ document.body.addEventListener("dragover", e => { e.preventDefault(); }); document.body.addEventListener("drop", e => { e.preventDefault(); const arquivo = e.dataTransfer.files[0]; if(!arquivo) return; if(!arquivo.type.startsWith("image")) return; const leitor = new FileReader(); leitor.onload = function(ev){ fotos.unshift({ imagem:ev.target.result, data:new Date().toLocaleString("pt-BR") }); salvarFotos(); mostrarFotos(); } leitor.readAsDataURL(arquivo); });
